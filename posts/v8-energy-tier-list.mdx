export function V8TierList() {
  const [selectedId, setSelectedId] = useState(FLAVORS[0].id)
  const selected = FLAVORS.find((f) => f.id === selectedId)
  const selectedTier = TIERS.find((t) => t.key === selected.tier)

  return (
    <div className="not-prose font-neo">
      <div className="border-4 border-black bg-white">
        {TIERS.map((tier) => {
          const cans = FLAVORS.filter((f) => f.tier === tier.key)
          return (
            <div key={tier.key} className="flex min-h-16 items-stretch border-b-4 border-black last:border-b-0 md:min-h-[4.5rem]">
              <div
                className={
                  'flex w-14 shrink-0 items-center justify-center border-r-4 border-black px-1 text-center font-mono font-extrabold leading-tight text-black break-words ' +
                  (tier.label.length <= 1 ? 'text-2xl' : 'text-sm')
                }
                style={{ backgroundColor: tier.color }}
              >
                {tier.label}
              </div>
              <div className="flex flex-wrap items-center gap-0.5 p-0.5">
                {cans.map((flavor) => {
                  const isSelected = flavor.id === selectedId
                  return (
                    <button
                      key={flavor.id}
                      type="button"
                      onClick={() => setSelectedId(flavor.id)}
                      aria-pressed={isSelected}
                      aria-label={flavor.name}
                      title={flavor.name}
                      className={
                        'border-2 transition-all duration-100 ' +
                        (isSelected
                          ? 'border-black bg-black'
                          : 'border-transparent bg-transparent hover:border-black')
                      }
                    >
                      <img
                        src={flavor.img}
                        alt={flavor.name}
                        loading="lazy"
                        className="h-14 w-14 object-contain md:h-16 md:w-16"
                      />
                    </button>
                  )
                })}
              </div>
            </div>
          )
        })}
      </div>

      <div className="mt-3 flex items-start gap-4 border-4 border-black bg-white p-3">
        <img
          src={selected.img}
          alt={selected.name}
          className="h-24 w-24 shrink-0 object-contain md:h-28 md:w-28"
        />
        <div>
          <div className="mb-1.5 flex items-center gap-2">
            <span
              className="flex h-7 w-7 shrink-0 items-center justify-center border-2 border-black font-mono text-base font-extrabold text-black"
              style={{ backgroundColor: selectedTier.color }}
            >
              {selectedTier.label}
            </span>
            <h3 className="font-neo text-xl font-extrabold uppercase tracking-tight text-black">
              {selected.name}
            </h3>
          </div>
          <p className="font-neo font-medium leading-relaxed text-black">
            {selected.review}
          </p>
        </div>
      </div>
    </div>
  )
}
