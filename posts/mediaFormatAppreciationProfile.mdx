import { useEffect, useRef, useState } from 'react'

I don't feel like different forms of media and how differently people relate to them gets enough attention.

The idea of reading a poem just doesn't resonate with me at all. I get that you're supposed to get excited about Mr. Nolan's Odyssey and travel far to see it in 70mm -- to me that sounds fun because it's an adventure and not because the format is that different. So I built this survey to encourage some self-reflection and made it enforce quantification because I think that's funny.

Rate your appreciation of each form **relative to everyone else** -- 50% means you figure you're about average, 100% means you get more out of it than anyone you know. Everyone's cards, mine first, are at the bottom.

{/*
  Editing is gated on a secret the server minted when the card was saved, and
  this browser is the only place it is ever kept. That is the whole ownership
  model: a share link lets you read someone's card and nothing else, so a card
  in the list below can only be rewritten by the person who wrote it.
*/}
export const TOKEN_STORE = 'mediaProfile:editTokens'

export function readTokens() {
  try {
    const parsed = JSON.parse(window.localStorage.getItem(TOKEN_STORE) ?? 'null')
    return parsed && typeof parsed === 'object' ? parsed : {}
  } catch {
    return {}
  }
}

export function rememberToken(id, token) {
  try {
    window.localStorage.setItem(TOKEN_STORE, JSON.stringify({ ...readTokens(), [id]: token }))
  } catch {
    // Private browsing, or storage that is full. The card is saved either way;
    // the only thing lost is the ability to come back and change it.
  }
}

{/* Rendered client-side only, so a locale-dependent date can't desync hydration. */}
export function formatStamp(result) {
  const stamp = result.updatedAt || result.savedAt
  if (!stamp) return ''
  const date = new Date(stamp)
  if (Number.isNaN(date.getTime())) return ''
  const text = date.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' })
  return result.updatedAt ? `Edited ${text}` : text
}

{/* A filled bar. The width has to be an inline style: Tailwind's JIT can't
    see a value that only exists at runtime. */}
export function Bar({ value, thin }) {
  return (
    <div className={`w-full border-2 border-black bg-white ${thin ? 'h-2' : 'h-4'}`}>
      <div className="h-full bg-black" style={{ width: `${value}%` }} />
    </div>
  )
}

export function ReadOnlyScale({ scale, result }) {
  const value = result.scores[scale.id]
  if (value === undefined) return null
  const note = result.notes?.[scale.id]
  const subs = (scale.subs ?? []).filter((sub) => result.scores[sub.id] !== undefined)

  return (
    <div className="border-b-2 border-black px-3 py-2.5 last:border-b-0">
      <div className="flex items-baseline justify-between gap-3">
        <span className="font-neo text-sm font-extrabold uppercase tracking-tight text-black">
          {scale.label}
        </span>
        <span className="font-mono text-sm font-bold tabular-nums text-black">{value}%</span>
      </div>
      <div className="mt-1.5">
        <Bar value={value} />
      </div>
      {subs.map((sub) => (
        <div key={sub.id} className="mt-2">
          <div className="flex items-baseline justify-between gap-3">
            <span className="font-neo text-xs font-medium text-black">{sub.label}</span>
            <span className="font-mono text-xs font-bold tabular-nums text-black">
              {result.scores[sub.id]}%
            </span>
          </div>
          <div className="mt-1">
            <Bar value={result.scores[sub.id]} thin />
          </div>
        </div>
      ))}
      {scale.ask && result.broadcaster !== undefined && (
        <p className="mt-1.5 pl-3 font-neo text-xs font-medium text-black">
          {scale.ask} {result.broadcaster ? 'Yes' : 'No'}
        </p>
      )}
      {note && (
        <p className="mt-2 border-l-4 border-black pl-2 font-neo text-sm font-medium leading-snug text-black">
          {note}
        </p>
      )}
    </div>
  )
}

{/*
  One person's answers. Collapsible in the list, where the title bands stack
  into the roster: closed, they are the whole list; open, one person's numbers
  hang off theirs. Without onToggle -- the card a share link opens -- it is
  always open and the band is inert.
*/}
export function ResultCard({ result, title, index, badge, meta, action, open, onToggle }) {
  const collapsible = typeof onToggle === 'function'
  const showBody = !collapsible || open

  return (
    <div className="not-prose border-4 border-black bg-white font-neo shadow-[6px_6px_0_0_#111]">
      <div role="heading" aria-level={3}>
        {collapsible ? (
          <button
            type="button"
            onClick={onToggle}
            aria-expanded={open}
            className={
              'flex w-full items-center bg-white px-3 py-2 text-left hover:bg-gray-100 ' +
              (showBody ? 'border-b-4 border-black' : '')
            }
          >
            <CardTitle title={title} index={index} badge={badge} toggle open={open} />
          </button>
        ) : (
          <div className="flex items-center border-b-4 border-black bg-white px-3 py-2">
            <CardTitle title={title} index={index} badge={badge} />
          </div>
        )}
      </div>
      {showBody && (meta || action) && (
        <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1 border-b-2 border-black bg-gray-100 px-3 py-1">
          <span className="font-mono text-[11px] font-medium uppercase tracking-wide text-black">
            {meta}
          </span>
          {action}
        </div>
      )}
      {showBody &&
        SCALES.map((scale) => <ReadOnlyScale key={scale.id} scale={scale} result={result} />)}
    </div>
  )
}

{/*
  A slider parked at 0 cannot report "0%" through onChange alone: the value
  never changes, so the browser fires nothing and the scale stays unrated.
  These handlers commit the current position the moment someone touches the
  control, which is the only way 0 is reachable now that 0 is also the
  resting position.
*/}
export function markRated(id, rated, onScore) {
  if (rated) return {}
  const commit = () => onScore(id, 0)
  return { onPointerDown: commit, onKeyDown: commit }
}

export function ScaleRow({ scale, scores, notes, broadcaster, onScore, onNote, onBroadcaster }) {
  const value = scores[scale.id]
  const rated = value !== undefined
  const [showNote, setShowNote] = useState(false)

  return (
    <div className="border-b-4 border-black px-3 py-3 last:border-b-0">
      <div className="flex flex-wrap items-baseline justify-between gap-x-3 gap-y-1">
        <label
          htmlFor={`scale-${scale.id}`}
          className="font-neo text-sm font-extrabold uppercase tracking-tight text-black"
        >
          {scale.label}
          {scale.hint && (
            <span className="ml-1.5 font-neo text-xs font-medium normal-case text-black opacity-60">
              ({scale.hint})
            </span>
          )}
        </label>
        <span className="font-mono text-sm font-bold tabular-nums text-black">
          {rated ? `${value}%` : '—'}
        </span>
      </div>

      <input
        id={`scale-${scale.id}`}
        type="range"
        min="0"
        max="100"
        step="10"
        value={rated ? value : 0}
        onChange={(e) => onScore(scale.id, Number(e.target.value))}
        {...markRated(scale.id, rated, onScore)}
        className="mt-2 h-2 w-full cursor-pointer appearance-none rounded-none border-2 border-black bg-gray-200 accent-black"
      />

      {rated && (
        <div className="mt-2">
          {(scale.subs ?? []).map((sub) => {
            const subValue = scores[sub.id]
            return (
              <div key={sub.id} className="mt-1.5 pl-3">
                <div className="flex flex-wrap items-baseline justify-between gap-x-3">
                  <label
                    htmlFor={`scale-${sub.id}`}
                    className="font-neo text-xs font-medium text-black"
                  >
                    {sub.label}
                  </label>
                  <span className="font-mono text-xs font-bold tabular-nums text-black">
                    {subValue !== undefined ? `${subValue}%` : '—'}
                  </span>
                </div>
                <input
                  id={`scale-${sub.id}`}
                  type="range"
                  min="0"
                  max="100"
                  step="10"
                  value={subValue !== undefined ? subValue : 0}
                  onChange={(e) => onScore(sub.id, Number(e.target.value))}
                  {...markRated(sub.id, subValue !== undefined, onScore)}
                  className="mt-1 h-1.5 w-full cursor-pointer appearance-none rounded-none border-2 border-black bg-gray-200 accent-black"
                />
              </div>
            )
          })}

          {scale.ask && (
            <div className="mt-2 flex flex-wrap items-center gap-2 pl-3">
              <span className="font-neo text-xs font-medium text-black">{scale.ask}</span>
              {[
                { label: 'Yes', value: true },
                { label: 'No', value: false },
              ].map((option) => (
                <button
                  key={option.label}
                  type="button"
                  onClick={() => onBroadcaster(broadcaster === option.value ? undefined : option.value)}
                  aria-pressed={broadcaster === option.value}
                  className={
                    'border-2 border-black px-2 py-0.5 font-neo text-xs font-extrabold uppercase ' +
                    (broadcaster === option.value
                      ? 'bg-black text-white'
                      : 'bg-white text-black hover:bg-black hover:text-white')
                  }
                >
                  {option.label}
                </button>
              ))}
            </div>
          )}

          {showNote || notes[scale.id] ? (
            <textarea
              value={notes[scale.id] ?? ''}
              onChange={(e) => onNote(scale.id, e.target.value)}
              maxLength={280}
              rows={2}
              placeholder="Optional note"
              aria-label={`Note about ${scale.label}`}
              className="mt-2 w-full border-2 border-black bg-white p-2 font-neo text-sm text-black placeholder:opacity-50"
            />
          ) : (
            <button
              type="button"
              onClick={() => setShowNote(true)}
              className="mt-2 font-neo text-xs font-extrabold uppercase tracking-tight text-black underline"
            >
              + Add a note
            </button>
          )}
        </div>
      )}
    </div>
  )
}

export function MediaProfile() {
  const [scores, setScores] = useState({})
  const [notes, setNotes] = useState({})
  const [broadcaster, setBroadcaster] = useState(undefined)
  const [name, setName] = useState('')
  const [isPublic, setIsPublic] = useState(false)
  // Set once this browser owns a card: after saving one, or after picking up
  // the edit token for one it saved earlier.
  const [owned, setOwned] = useState(null)
  const [shareUrl, setShareUrl] = useState('')
  const [phase, setPhase] = useState('editing')
  const [error, setError] = useState('')
  const [shared, setShared] = useState(null)
  const [copied, setCopied] = useState(false)
  const formRef = useRef(null)

  function linkFor(id) {
    return `${window.location.origin}${window.location.pathname}?r=${id}`
  }

  // Load the card the form should be editing, replacing whatever is in it.
  function beginEdit(id, token, result) {
    setScores(result.scores ?? {})
    setNotes(result.notes ?? {})
    setBroadcaster(result.broadcaster)
    setName(result.name ?? '')
    setIsPublic(Boolean(result.isPublic))
    setOwned({ id, token })
    setShareUrl(linkFor(id))
    setShared(null)
    setPhase('editing')
    setError('')
  }

  // Read ?r= on the client only. Doing it during render would make the server
  // and the first client pass disagree and blow up hydration.
  useEffect(() => {
    const id = new URLSearchParams(window.location.search).get('r')
    if (!id) return
    setShared({ state: 'loading' })
    fetch(`/api/mediaProfile/share?id=${encodeURIComponent(id)}`)
      .then((res) => (res.ok ? res.json() : Promise.reject(new Error('not found'))))
      .then((data) => setShared({ state: 'ready', id, result: data.result, mine: Boolean(readTokens()[id]) }))
      .catch(() => setShared({ state: 'missing' }))
  }, [])

  // The list at the bottom asks for a card back when its author clicks Edit.
  useEffect(() => {
    function onEditRequest(event) {
      const id = event.detail?.id
      const token = id ? readTokens()[id] : undefined
      if (!token) return
      fetch(`/api/mediaProfile/share?id=${encodeURIComponent(id)}`)
        .then((res) => (res.ok ? res.json() : Promise.reject(new Error('gone'))))
        .then((data) => {
          beginEdit(id, token, data.result)
          formRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
        })
        .catch(() => setError('Could not load that card to edit.'))
    }
    window.addEventListener(EDIT_EVENT, onEditRequest)
    return () => window.removeEventListener(EDIT_EVENT, onEditRequest)
  }, [])

  const rated = Object.keys(scores).filter((id) => !id.includes('.')).length
  const editing = owned !== null

  function handleScore(id, value) {
    setScores((prev) => ({ ...prev, [id]: value }))
  }

  function handleNote(id, value) {
    setNotes((prev) => ({ ...prev, [id]: value }))
  }

  async function handleSave() {
    setPhase('saving')
    setError('')
    try {
      const payload = {
        scores,
        notes: Object.fromEntries(Object.entries(notes).filter(([, v]) => v.trim())),
        ...(broadcaster === undefined ? {} : { broadcaster }),
        ...(name.trim() ? { name: name.trim() } : {}),
        isPublic,
        ...(editing ? { id: owned.id, editToken: owned.token } : {}),
      }
      const res = await fetch('/api/mediaProfile/share', {
        method: editing ? 'PUT' : 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data.error || 'Could not save that.')

      if (!editing) {
        rememberToken(data.id, data.editToken)
        setOwned({ id: data.id, token: data.editToken })
        setShareUrl(linkFor(data.id))
      }
      setPhase('done')
      // Publishing, un-publishing and editing all change what the list shows.
      window.dispatchEvent(
        new CustomEvent(PUBLISHED_EVENT, { detail: { id: editing ? owned.id : data.id } })
      )
    } catch (e) {
      setError(e.message)
      setPhase('editing')
    }
  }

  async function handleCopy() {
    try {
      await navigator.clipboard.writeText(shareUrl)
      setCopied(true)
      setTimeout(() => setCopied(false), 2000)
    } catch {
      setCopied(false)
    }
  }

  return (
    <div ref={formRef} className="not-prose mb-8 scroll-mt-6 font-neo">
      {shared?.state === 'ready' && (
        <div className="mb-8">
          <ResultCard
            result={shared.result}
            title={shared.result.name ? `${shared.result.name}'s card` : 'A shared card'}
            badge={shared.mine ? 'Yours' : undefined}
            meta={formatStamp(shared.result)}
            action={
              shared.mine ? (
                <button
                  type="button"
                  onClick={() => beginEdit(shared.id, readTokens()[shared.id], shared.result)}
                  className="border-2 border-black bg-white px-2 py-0.5 font-neo text-[11px] font-extrabold uppercase tracking-tight text-black hover:bg-black hover:text-white"
                >
                  Edit
                </button>
              ) : undefined
            }
          />
          <p className="mt-2 font-neo text-sm font-medium text-black">
            {shared.mine ? 'This is your card. Fill it in again below, or edit it.' : 'Fill out your own below.'}
          </p>
        </div>
      )}
      {shared?.state === 'missing' && (
        <p className="mb-6 border-4 border-black bg-white p-3 font-neo text-sm font-medium text-black">
          That shared card has expired or never existed. You can still fill out your own.
        </p>
      )}

      {editing && phase !== 'done' && (
        <p className="mb-3 border-4 border-black bg-black p-3 font-neo text-sm font-extrabold uppercase tracking-tight text-white">
          Editing your card — saving replaces it at the same link.
        </p>
      )}

      <div className="border-4 border-black bg-white">
        {SCALES.map((scale) => (
          <ScaleRow
            key={scale.id}
            scale={scale}
            scores={scores}
            notes={notes}
            broadcaster={broadcaster}
            onScore={handleScore}
            onNote={handleNote}
            onBroadcaster={setBroadcaster}
          />
        ))}
      </div>

      <div className="mt-3 border-4 border-black bg-white p-3">
        {phase !== 'done' && (
          <>
            <div className="mt-2">
              <label htmlFor="profile-name" className="font-neo text-xs font-extrabold uppercase text-black">
                Your name <span className="font-medium normal-case opacity-60">(optional)</span>
              </label>
              <input
                id="profile-name"
                type="text"
                value={name}
                onChange={(e) => setName(e.target.value)}
                maxLength={40}
                placeholder="Goes on the card you share"
                className="mt-1 w-full border-2 border-black bg-white p-2 font-neo text-sm text-black placeholder:opacity-50"
              />
            </div>

            <button
              type="button"
              onClick={() => setIsPublic((prev) => !prev)}
              aria-pressed={isPublic}
              className="mt-3 flex w-full items-start gap-2 border-2 border-black bg-white p-2 text-left hover:bg-gray-100"
            >
              <span
                aria-hidden="true"
                className={
                  'mt-0.5 flex h-4 w-4 shrink-0 items-center justify-center border-2 border-black font-mono text-[10px] font-bold leading-none ' +
                  (isPublic ? 'bg-black text-white' : 'bg-white text-transparent')
                }
              >
                X
              </span>
              <span className="font-neo text-xs font-extrabold uppercase tracking-tight text-black">
                Show my card in the responses at the bottom
              </span>
            </button>

            <button
              type="button"
              onClick={handleSave}
              disabled={rated === 0 || phase === 'saving'}
              className="mt-3 w-full border-4 border-black bg-white px-4 py-2 font-neo text-sm font-extrabold uppercase tracking-tight text-black hover:bg-black hover:text-white disabled:cursor-not-allowed disabled:opacity-40 disabled:hover:bg-white disabled:hover:text-black"
            >
              {phase === 'saving' ? 'Saving…' : editing ? 'Save changes' : 'Share my card'}
            </button>
          </>
        )}

        {error && (
          <p role="alert" className="mt-2 font-neo text-sm font-bold text-black">
            {error}
          </p>
        )}

        {phase === 'done' && (
          <div className="mt-2">
            <label htmlFor="share-url" className="font-neo text-xs font-extrabold uppercase text-black">
              Your link
            </label>
            <div className="mt-1 flex flex-col gap-2 sm:flex-row">
              <input
                id="share-url"
                type="text"
                readOnly
                value={shareUrl}
                onFocus={(e) => e.target.select()}
                className="w-full border-2 border-black bg-white p-2 font-mono text-xs text-black"
              />
              <button
                type="button"
                onClick={handleCopy}
                className="shrink-0 border-4 border-black bg-white px-4 py-2 font-neo text-xs font-extrabold uppercase tracking-tight text-black hover:bg-black hover:text-white"
              >
                {copied ? 'Copied' : 'Copy'}
              </button>
            </div>
            <p className="mt-2 font-neo text-sm font-medium text-black">
              {isPublic
                ? 'It is in the responses at the bottom of the post.'
                : 'Only people you send the link to can see it.'}
            </p>
            <button
              type="button"
              onClick={() => setPhase('editing')}
              className="mt-2 w-full border-4 border-black bg-white px-4 py-2 font-neo text-sm font-extrabold uppercase tracking-tight text-black hover:bg-black hover:text-white"
            >
              Edit my card
            </button>
            <p className="mt-2 font-neo text-xs font-medium text-black opacity-60">
              Editing works from this browser only — the key that proves the card is yours is
              stored here and nowhere else.
            </p>
          </div>
        )}
      </div>
    </div>
  )
}

export function nameFor(card) {
  return card.result.name || 'Anonymous'
}

export function ResponsesList() {
  // Which cards are open. Nothing is, to start: the point of the list is to
  // compare yourself to the population, and you pick your own numbers first.
  const [openIds, setOpenIds] = useState([])
  // Zero on a first look; the moment of the save that asked for a re-read
  // otherwise. It doubles as the cache-buster: the endpoint lets a plain
  // listing sit in the browser cache for a minute, which is a minute too long
  // for somebody waiting to see the edit they just made.
  const [refreshedAt, setRefreshedAt] = useState(0)
  const [state, setState] = useState({ status: 'loading', cards: [] })
  const [mine, setMine] = useState({})

  function toggle(key) {
    setOpenIds((prev) => (prev.includes(key) ? prev.filter((k) => k !== key) : [...prev, key]))
  }

  useEffect(() => {
    setMine(readTokens())
    function onPublished(event) {
      setRefreshedAt(Date.now())
      setMine(readTokens())
      // Open the card that was just saved, so the change is visible rather
      // than filed away behind a row you have to find.
      const id = event.detail?.id
      if (id) setOpenIds((prev) => (prev.includes(id) ? prev : [...prev, id]))
    }
    window.addEventListener(PUBLISHED_EVENT, onPublished)
    return () => window.removeEventListener(PUBLISHED_EVENT, onPublished)
  }, [])

  useEffect(() => {
    let cancelled = false
    fetch(refreshedAt ? `/api/mediaProfile/public?t=${refreshedAt}` : '/api/mediaProfile/public')
      .then((res) => (res.ok ? res.json() : Promise.reject(new Error('failed'))))
      .then((data) => {
        if (!cancelled) setState({ status: 'ready', cards: data.cards ?? [] })
      })
      .catch(() => {
        if (!cancelled) setState((prev) => ({ status: 'error', cards: prev.cards }))
      })
    return () => {
      cancelled = true
    }
  }, [refreshedAt])

  const entries = [
    { key: 'tyler', title: 'Tyler', result: TYLERS_CARD, badge: 'Author' },
    ...state.cards.map((card) => ({
      key: card.id,
      id: card.id,
      title: nameFor(card),
      result: card.result,
      badge: mine[card.id] ? 'Yours' : undefined,
      meta: formatStamp(card.result),
    })),
  ]

  return (
    <div className="not-prose font-neo">
      <ol className="list-none space-y-4 pl-0">
        {entries.map((entry, position) => (
          // The card prints its own number in its title band, and the post
          // body is `prose`, whose ordered-list numbering would sit a second
          // one out in the margin. Inline because it has to beat prose.
          <li key={entry.key} style={{ listStyle: 'none' }}>
            <ResultCard
              result={entry.result}
              title={entry.title}
              index={position + 1}
              badge={entry.badge}
              meta={entry.meta}
              open={openIds.includes(entry.key)}
              onToggle={() => toggle(entry.key)}
              action={
                entry.id && mine[entry.id] ? (
                  <button
                    type="button"
                    onClick={() =>
                      window.dispatchEvent(new CustomEvent(EDIT_EVENT, { detail: { id: entry.id } }))
                    }
                    className="border-2 border-black bg-white px-2 py-0.5 font-neo text-[11px] font-extrabold uppercase tracking-tight text-black hover:bg-black hover:text-white"
                  >
                    Edit
                  </button>
                ) : undefined
              }
            />
          </li>
        ))}
      </ol>

      {state.status === 'error' && (
        <p className="mt-6 border-4 border-black bg-white p-3 font-neo text-sm font-medium text-black">
          Could not load the responses right now.
        </p>
      )}
    </div>
  )
}
